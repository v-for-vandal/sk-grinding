cmake_minimum_required(VERSION 3.25)
project(SkillPractice VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(tomlplusplus CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(PracticeCore STATIC src/Core.cpp src/Config.cpp)
target_include_directories(PracticeCore PUBLIC src)
file(GLOB_RECURSE DEFAULT_CONFIGS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../config/*.toml")
# Compiled copy of config/*.toml (fallback and source of missing keys). Stored
# as byte arrays: MSVC limits one string literal to ~16 KB (C2026).
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${DEFAULT_CONFIGS})
set(DEFAULT_HEADER "#pragma once\n#include <map>\n#include <string>\nnamespace sp {\nnamespace defaults_data {\n")
set(DEFAULT_ENTRIES "")
set(DEFAULT_INDEX 0)
foreach(CONFIG_FILE IN LISTS DEFAULT_CONFIGS)
  file(RELATIVE_PATH CONFIG_NAME "${CMAKE_CURRENT_SOURCE_DIR}/../config" "${CONFIG_FILE}")
  file(READ "${CONFIG_FILE}" CONFIG_HEX HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," CONFIG_BYTES "${CONFIG_HEX}")
  string(APPEND DEFAULT_HEADER "inline constexpr unsigned char f${DEFAULT_INDEX}[] = {${CONFIG_BYTES}0};\n")
  string(APPEND DEFAULT_ENTRIES "{\"${CONFIG_NAME}\", std::string(reinterpret_cast<const char*>(defaults_data::f${DEFAULT_INDEX}), sizeof(defaults_data::f${DEFAULT_INDEX}) - 1)},\n")
  math(EXPR DEFAULT_INDEX "${DEFAULT_INDEX} + 1")
endforeach()
string(APPEND DEFAULT_HEADER "}\ninline const std::map<std::string, std::string> defaults{\n${DEFAULT_ENTRIES}};\n}  // namespace sp\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Defaults.h" CONTENT "${DEFAULT_HEADER}" @ONLY)
target_include_directories(PracticeCore PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(PracticeCore PUBLIC tomlplusplus::tomlplusplus nlohmann_json::nlohmann_json)
enable_testing()
add_executable(PracticeTests tests/CoreTests.cpp)
target_link_libraries(PracticeTests PRIVATE PracticeCore)
add_test(NAME PracticeCore COMMAND PracticeTests "${CMAKE_CURRENT_SOURCE_DIR}/../config")
option(PRACTICE_PLUGIN "Build the SKSE DLL" ON)
if(PRACTICE_PLUGIN)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
  set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
  add_subdirectory(extern/CommonLibSSE-NG)
  target_compile_options(CommonLibSSE PRIVATE /MP4)
  include(extern/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
  find_package(minhook CONFIG REQUIRED)
  add_commonlibsse_plugin(SkillPractice NAME SkillPractice AUTHOR "Skill Practice" VERSION 0.1.4 USE_ADDRESS_LIBRARY SOURCES src/Plugin.cpp src/State.cpp src/Telemetry.cpp src/Classification.cpp src/UI.cpp src/Aim.cpp)
  target_link_libraries(SkillPractice PRIVATE PracticeCore minhook::minhook)
  target_include_directories(SkillPractice PRIVATE extern)
  target_compile_definitions(SkillPractice PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_compile_options(SkillPractice PRIVATE /Zi)
  target_link_options(SkillPractice PRIVATE /DEBUG:FULL /OPT:REF)
  target_precompile_headers(SkillPractice PRIVATE src/PCH.h)
endif()
